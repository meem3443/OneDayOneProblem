### **내 코드 (초안)**

**코드 스니펫**

```java
import java.io.*;

import java.util.*;

  

public class Main {

static int N;

public static void main(String[] args) throws IOException{

BufferedReader br = new BufferedReader(new InputStreamReader(System.in));

StringBuilder sb = new StringBuilder();

N = Integer.parseInt(br.readLine());

int num[][] = new int[N][N];

int max = -1;

  

for(int i = 0; i < N; i++){

StringTokenizer st = new StringTokenizer(br.readLine());

for(int j = 0; j < N; j++){

num[i][j] = Integer.parseInt(st.nextToken());

}

}

  

for(int j = 1; j < N-1; j++){

int x = N-1;

int y = j;

max = Math.max(max, sumOfLine(num, x, y));

}

if(N%2==0) max = Math.max(max, sumOfMid(num));

System.out.println(max);

  
  

}

public static int sumOfLine(int arr[][], int x, int y){

int result = 0;

int i= x;

for(int j = y; j >= 0; j--){

result += arr[i][j];

result += arr[j][i];

i--;

}

i = x-1;

for(int j = y+1; j < N-1; j++){

result += arr[i][j];

result += arr[j][i];

i--;

}

return result;

}

public static int sumOfMid(int arr[][]){

int x = N/2;

int y = N/2;

int result1 = 0;//오른쪽

int result2 = 0;//왼쪽

int Ly, Ry;

int i = x; int j = y;

while(i > 0 && j > 0){

result1 += arr[i][j];

i--; j--;

}

i--; j++;

while(i!=0){

result1 += arr[i][j];

i--; j++;

}

Ly = j;

for(int k = 1; k <= x-1; k++){

result1 += arr[i+k][j+k];

}

for(int k = 1; k < Ly-1; k++){

result1 += arr[x-k][y+k];

}

i = x; i = y;

while(i < N-1 && j < N-1){

result2 += arr[i][j];

i--; j++;

}

i--; j--;

while(i!=0){

result2 += arr[i][j];

i--; j--;

}

Ry = j;

for(int k = 1; k <=x-1; k++){

result2 += arr[i+k][i-k];

}

for(int k = 1; k < N-2-Ry; k++){

result2 += arr[x-k][y-k];

}

int result = Math.max(result1, result2);

return result;

}

  

}
```

### **AI 최적화 코드 비교 및 진단**
```java
import java.util.Scanner;

public class Main {
    public static final int DIR_NUM = 4;
    public static final int MAX_N = 20;
    
    public static int n;
    public static int[][] grid = new int[MAX_N][MAX_N];
    
    public static boolean inRange(int x, int y) {
        return 0 <= x && x < n && 0 <= y && y < n;
    }
    
    public static int getScore(int x, int y, int k, int l) {
        int[] dx = new int[]{-1, -1, 1, 1};
        int[] dy = new int[]{1, -1, -1, 1};
        int[] moveNum = new int[]{k, l, k, l};
        
        int sumOfNums = 0;
    
        // 기울어진 직사각형의 경계를 쭉 따라가봅니다.
        for(int d = 0; d < DIR_NUM; d++)
            for(int q = 0; q < moveNum[d]; q++) {
                x += dx[d]; y += dy[d];
                    
                // 기울어진 직사각형이 경계를 벗어나는 경우라면
                // 불가능하다는 의미로 답이 갱신되지 않도록
                // 0을 반환합니다.
                if(!inRange(x, y))
                    return 0;
                
                sumOfNums += grid[x][y];
            }
        
        return sumOfNums;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        n = sc.nextInt();
        
        for(int i = 0; i < n; i++)
            for(int j = 0; j < n; j++)
                grid[i][j] = sc.nextInt();
        
        int ans = 0;
        
        // (i, j)를 시작으로 1, 2, 3, 4 방향
        // 순서대로 길이 [k, l, k, l] 만큼 이동하면 그려지는
        // 기울어진 직사각형을 잡아보는
        // 완전탐색을 진행해봅니다.
        for(int i = 0; i < n; i++)
            for(int j = 0; j < n; j++)
                for(int k = 1; k < n; k++)
                    for(int l = 1; l < n; l++)
                        ans = Math.max(ans, getScore(i, j, k, l));

        System.out.print(ans);
    }
}
```
