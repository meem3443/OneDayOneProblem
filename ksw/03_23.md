### **내 코드 (초안)**

**코드 스니펫**

```java
import java.io.*;
import java.util.*;

public class Main {
    public static void main(String[] args) throws IOException{
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        StringTokenizer st = new StringTokenizer(br.readLine());
        int size = Integer.parseInt(st.nextToken());
        int price = Integer.parseInt(st.nextToken());
        int cost;
        int result = 0;
        int[][] goldMine = new int[size][size];
        int sum = 0;

        //금광산 배열 입력
        for(int i = 0; i < size; i++){
            st = new StringTokenizer(br.readLine());
            for(int j = 0; j < size; j++){
                goldMine[i][j] = Integer.parseInt(st.nextToken());
                sum+=goldMine[i][j];
            }
        }

        //마름모형태 내의 금의 개수 비교 by manhattan distance
        for(int k = 0; k <= size; k++){
            cost = k*k + (k+1)*(k+1);
            
            for(int x = 0; x < size; x++){
                for(int y = 0; y < size; y++){

                    int countGold = 0;

                    for(int i = 0; i < size; i++){
                        for(int j = 0; j < size; j++){
                            if((Math.abs(i-x)+Math.abs(j-y))<=k)
                                countGold += goldMine[i][j];
                        }
                    }
                    if(countGold*price>=cost) result = Math.max(result, countGold);
            
                }
            }
            
        }
        if(sum>result && sum*price>=size*size+(size+1)*(size+1)) result = sum;
        System.out.println(result);


        

        
    }

    
}
```
